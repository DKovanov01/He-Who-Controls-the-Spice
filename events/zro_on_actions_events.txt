country_event = {
	id = zro_worm.13
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		from = {
			OR = {
				has_ruler_trait = leader_trait_ruler_prescience
			}
			NOR = {
				has_trait = leader_trait_admiral_prescience
				has_trait = leader_trait_general_prescience
				has_trait = leader_trait_governor_prescience
				has_trait = leader_trait_scientist_prescience
			}
		}
	}

	immediate = {
		from = {
			if = {
				limit = { leader_class = admiral }
				add_trait = leader_trait_admiral_prescience
				break = yes
			}
			if = {
				limit = { leader_class = general }
				add_trait = leader_trait_general_prescience
				break = yes
			}
			if = {
				limit = { leader_class = governor }
				add_trait = leader_trait_governor_prescience
				break = yes
			}
			if = {
				limit = { leader_class = scientist }
				add_trait = leader_trait_scientist_prescience
				break = yes
			}
		}
	}
}

country_event = {
	id = zro_worm.12
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		from = {
			OR = {
				has_trait = leader_trait_admiral_mentat
				has_trait = leader_trait_general_mentat
				has_trait = leader_trait_governor_mentat
				has_trait = leader_trait_scientist_mentat
			}
			NOT = {
			    has_ruler_trait = trait_ruler_mentat
			}
		}
	}

	immediate = {
		from = {
				limit = { leader_class = ruler }
				add_trait = trait_ruler_mentat
				break = yes
		}
	}
}
