#REGULAR MEGACORP#
building_zro_smugglers_legit = {
	base_buildtime = @b1_time
	base_cap_amount = 3
	icon = building_zro_harvester
	
	owner_type = corporate

	potential = {
		has_branch_office = yes
		is_artificial = no
		branch_office_owner = {
			is_criminal_syndicate = no
		}
		NOR = {
			is_planet_class = pc_hive
			is_planet_class = pc_machine
			is_planet_class = pc_city
		}
		OR = { 
			    has_deposit = d_zro_worm_home
				has_deposit = d_zro_great_sandworm
				}
	}

	planet_modifier = {
		trade_value_add = 25
		planet_crime_add = 10
	}

	resources = {
		category = planet_branch_office_buildings
		cost = {
			minerals = 500
			alloys = 100
		}
		produces = {
			sr_zro = 2
		}
		upkeep = {
			energy = 2
			alloys = 2
		}
	}

	ai_weight = {
		weight = 1500
		modifier = {
			factor = 3
			branch_office_owner = {
				has_monthly_income = {
					resource = sr_zro
					value < 10000
				}
			}
		}
	}
}
