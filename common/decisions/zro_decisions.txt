decision_great_sandworm_terraform = {
	sound = event_administrative_work
	icon = decision_consecrated_worlds
	owned_planets_only = yes
	
	potential = {
		exists = space_owner
		exists = from
		space_owner = { is_same_value = from }
		space_owner = { has_country_flag = relic_great_sandworm_activated }
		NOT = { has_planet_flag = zro_worm_terraforming_flag }
		NOT = { has_planet_flag = zro_great_worm_terraforming_flag }
		NOT = {
		has_deposit = d_zro_worm_colony
	    has_deposit = d_zro_worm_home
		has_deposit = d_zro_great_sandworm 
		is_planet_class = pc_city
		is_planet_class = pc_machine 
		is_planet_class = pc_relic 
		}
		OR = {
		has_modifier = terraforming_candidate
		is_colonizable = yes
		is_colony = yes
		}
		#NOR = {
		#	has_planet_flag = consecrated@from
		#	has_modifier = consecrated_quality_1
		#	has_modifier = consecrated_quality_2
		#	has_modifier = consecrated_quality_3
		#	has_modifier = consecrated_quality_4
		#}
	}
	
	resources = {
		category = decisions
		cost = {
		trigger = {
				OR = { 
				is_colonizable = yes
				is_colony = yes
				}
				NOT = { is_planet_class = pc_desert }
			}
			energy = 5000
			influence = 250
		}
		cost = {
		trigger = {
				OR = {
				is_colonizable = yes
				is_colony = yes
				}
				is_planet_class = pc_desert
			}
			energy = 2500
			influence = 250
		}
		cost = {
		trigger = {
				is_colony = yes
			}
			energy = 1000
			influence = 250
		}
	}


	effect = {
		if = {
	 	    limit = {
	 	        OR = {
	 	        is_colonizable = yes
	 	        is_colony = yes
	 	        }
	 	    }
	 	    add_modifier = { modifier = "planet_great_sandworm_terraforming" days = 1800 }
		}
		if = {
		    limit = {
		    is_colonizable = no
		    is_colony = no
		    }
		    add_modifier = { modifier = "planet_zro_worm_terraforming" }
		    remove modifier = terraforming_candidate
		}
		set_planet_flag = zro_great_sandworm_terraforming_flag
		custom_tooltip = zro_great_sandworm.terraform_tooltip
		ROOT = {
			hidden_effect = {
				planet_event = { id = zro_worm.5 days = 1800 }
			}
		}
		if = {
	 		limit = {
	 		    NOT = { is_planet_class = pc_desert }
	 		}
		}
		ROOT = {
			hidden_effect = {
				planet_event = { id = terra.5 days = 1 }
			}
		}
		hidden_effect = {
		    space_owner = { country_event = { id = zro_worm.6 days = 1800 } }
			owner = { remove_country_flag = relic_great_sandworm_activated }
			#planet_event = { id = zro_worm.18 } #Notification event
		}
	}

	ai_weight = {
		weight = 100
		modifier = {
			factor = 10
			owner = { has_origin = origin_zro_worm }
		}
		modifier = {
		    factor = 5
			has_active_building = building_zro_dispensery
	    }
		modifier = {
			factor = 2
			owner = {
				has_monthly_income = {
					resource = sr_zro
					value < 100
				}
			}
		}
	}
	
}
