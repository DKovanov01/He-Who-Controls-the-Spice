tech_zro = {
	cost = @tier2cost1
	area = society
	tier = 2
	category = { psionics }
	prerequisites = { "tech_basic_industry" }
	weight = @tier2weight1
	
	modifier = {
	    show_only_custom_tooltip = yes
		custom_tooltip = tech_zro_effect
		}

	weight_modifier = {
		factor = 5
		modifier = {
			factor = 5
			OR = {
				any_owned_planet = {
					OR = {
						has_deposit = d_zro_worm_home
						has_deposit = d_zro_great_sandworm
						has_deposit = d_zro_worm_colony
					}
				}
				any_planet_within_border = {
					OR = {
						has_deposit = d_zro_deposit_1
						has_deposit = d_zro_deposit_2
						has_deposit = d_zro_deposit_3
						has_deposit = d_zro_deposit_4
						has_deposit = d_zro_deposit_5
					}
				}
				any_owned_planet = {
					OR = {
						has_building = building_zro_dispensery 
					}
				}
			}
		}
		modifier = {
			factor = value:tech_weight_likelihood
			research_leader = {
				area = physics
				has_trait = "leader_trait_expertise_psionics"
			}
		}
	}

	ai_weight = {
		factor = 1000 #important resource
		modifier = {
			factor = 10
			research_leader = {
				area = physics
				has_trait = "leader_trait_expertise_psionics"
			}
		}
	}
}
